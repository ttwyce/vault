## 2. 建表脚本

### 2.1 核心

```sql
CREATE TABLE entry (
  id            TEXT PRIMARY KEY,          -- UUIDv7（时间有序）
  kind          INTEGER NOT NULL,          -- 1笔记 2想法 3日记 4AI日记 5书签 6密码
  title         TEXT,
  body          TEXT,                      -- Markdown
  meta          TEXT,                      -- JSON，见 §3
  created_at    INTEGER NOT NULL,          -- UTC ms
  updated_at    INTEGER NOT NULL,
  hlc           TEXT NOT NULL,             -- "0000018F2A3B4C5D:0003:devA"
  rev           INTEGER NOT NULL DEFAULT 1,
  content_hash  TEXT NOT NULL,             -- sha256(canonical)
  deleted       INTEGER NOT NULL DEFAULT 0,
  deleted_at    INTEGER
);
CREATE INDEX idx_entry_kind_time ON entry(kind, deleted, updated_at DESC);
CREATE INDEX idx_entry_hlc       ON entry(hlc);

CREATE TABLE entry_history (
  entry_id   TEXT NOT NULL,
  rev        INTEGER NOT NULL,
  payload    BLOB NOT NULL,                -- 加密快照
  hlc        TEXT NOT NULL,
  created_at INTEGER NOT NULL,
  PRIMARY KEY (entry_id, rev)
);

CREATE TABLE trash (
  entry_id   TEXT PRIMARY KEY,
  payload    BLOB NOT NULL,
  deleted_at INTEGER NOT NULL,
  expire_at  INTEGER NOT NULL              -- 默认 +30 天
);
```

### 2.2 组织：分类与标签

```sql
CREATE TABLE category (
  id          TEXT PRIMARY KEY,
  kind        INTEGER NOT NULL,            -- 归属类型，与 entry.kind 对齐
  parent_id   TEXT,                        -- NULL=一级；非空=二级（禁止三级）
  name        TEXT NOT NULL,
  color       TEXT,
  icon        TEXT,
  sort_order  INTEGER NOT NULL DEFAULT 0,
  entry_count INTEGER NOT NULL DEFAULT 0,  -- 冗余计数，异步更新
  created_at  INTEGER, updated_at INTEGER,
  hlc         TEXT NOT NULL,
  deleted     INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX idx_cat_kind ON category(kind, parent_id, sort_order);

CREATE TABLE tag (
  id          TEXT PRIMARY KEY,
  name        TEXT NOT NULL,
  color       TEXT,
  usage_count INTEGER NOT NULL DEFAULT 0,
  created_at  INTEGER, updated_at INTEGER,
  hlc         TEXT NOT NULL,
  deleted     INTEGER NOT NULL DEFAULT 0
);
CREATE UNIQUE INDEX idx_tag_name ON tag(name) WHERE deleted = 0;

CREATE TABLE entry_category (
  entry_id    TEXT NOT NULL,
  category_id TEXT NOT NULL,
  PRIMARY KEY (entry_id, category_id)
);
CREATE TABLE entry_tag (
  entry_id TEXT NOT NULL,
  tag_id   TEXT NOT NULL,
  PRIMARY KEY (entry_id, tag_id)
);

CREATE TABLE category_field_template (
  category_id TEXT NOT NULL,
  label       TEXT NOT NULL,
  type        TEXT NOT NULL DEFAULT 'text',  -- text|url|email|phone|date|multiline
  secret      INTEGER NOT NULL DEFAULT 0,
  sort_order  INTEGER NOT NULL DEFAULT 0,
  PRIMARY KEY (category_id, label)
);
```

### 2.3 账号

```sql
CREATE TABLE credential (
  entry_id            TEXT PRIMARY KEY,    -- 一对一 entry(kind=6)
  category_id         TEXT,
  username            TEXT,                -- 加密
  password            TEXT,                -- 加密
  totp_secret         TEXT,                -- 加密
  url                 TEXT,
  notes               TEXT,                -- 加密，自由备注
  custom_fields       TEXT,                -- 加密 JSON，见 §4
  password_updated_at INTEGER,
  strength            INTEGER,             -- 0~4，本地估算
  reused              INTEGER DEFAULT 0,
  leaked              INTEGER DEFAULT 0,
  favorite            INTEGER DEFAULT 0,
  icon_blob           TEXT                 -- favicon 的 blob hash
);
```

### 2.4 会话

```sql
CREATE TABLE conversation (
  id            TEXT PRIMARY KEY,
  title         TEXT,                      -- 默认取首条消息前 20 字
  group_id      TEXT,                      -- 可空；空则按日期自动分组
  created_at    INTEGER,
  updated_at    INTEGER,
  hlc           TEXT NOT NULL,
  message_count INTEGER DEFAULT 0,
  deleted       INTEGER DEFAULT 0
);
CREATE INDEX idx_conv_time ON conversation(deleted, updated_at DESC);

CREATE TABLE message (
  id              TEXT PRIMARY KEY,
  conversation_id TEXT NOT NULL,
  role            INTEGER NOT NULL,        -- 1 user / 2 assistant / 3 tool
  content         TEXT,
  tool_calls      TEXT,                    -- JSON
  citations       TEXT,                    -- JSON，entry id 数组
  created_at      INTEGER,
  hlc             TEXT NOT NULL
);
CREATE INDEX idx_msg_conv ON message(conversation_id, created_at);
```

### 2.5 日志与同步

```sql
CREATE TABLE oplog (
  seq        INTEGER PRIMARY KEY AUTOINCREMENT,
  hlc        TEXT NOT NULL,
  device_id  TEXT NOT NULL,
  entity     TEXT NOT NULL,                -- entry|category|tag|credential|conversation|message|blob|meta
  entity_id  TEXT NOT NULL,
  op         INTEGER NOT NULL,             -- 1 create 2 update 3 delete 4 restore 5 merge
  payload    BLOB,
  base_rev   INTEGER,
  applied    INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL
);
CREATE INDEX idx_oplog_pending ON oplog(applied, seq);

CREATE TABLE sync_state (
  path TEXT PRIMARY KEY, etag TEXT, size INTEGER, hlc TEXT, updated_at INTEGER
);
CREATE TABLE device_watermark (
  device_id TEXT PRIMARY KEY, max_hlc TEXT, last_seen INTEGER,
  platform TEXT, app_version TEXT
);
CREATE TABLE sync_log (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  started_at INTEGER, finished_at INTEGER, trigger TEXT, result TEXT,
  pulled INTEGER, pushed INTEGER, conflicts INTEGER,
  error_code TEXT, error_msg TEXT, request_count INTEGER
);
```

### 2.6 索引与检索

```sql
CREATE VIRTUAL TABLE entry_fts USING fts5(
  entry_id UNINDEXED, title, body, tags,
  tokenize = 'unicode61 remove_diacritics 2'
);
CREATE VIRTUAL TABLE conversation_fts USING fts5(
  conversation_id UNINDEXED, title, body,
  tokenize = 'unicode61 remove_diacritics 2'
);

CREATE TABLE diary_index (
  date          TEXT PRIMARY KEY,          -- YYYY-MM-DD（本地时区）
  user_entry_id TEXT,                      -- kind=3
  ai_entry_id   TEXT,                      -- kind=4
  word_count    INTEGER DEFAULT 0,
  has_mood      INTEGER DEFAULT 0
);

CREATE TABLE entry_vec (
  entry_id    TEXT NOT NULL,
  chunk_idx   INTEGER NOT NULL DEFAULT 0,  -- 长文分块
  model_id    TEXT NOT NULL,
  dim         INTEGER NOT NULL,
  vec         BLOB NOT NULL,               -- int8 量化
  computed_at INTEGER NOT NULL,
  PRIMARY KEY (entry_id, chunk_idx)
);
CREATE INDEX idx_vec_model ON entry_vec(model_id);

CREATE TABLE embedding_model (
  id TEXT PRIMARY KEY, family TEXT, dim INTEGER, mrl TEXT, ctx INTEGER,
  quant TEXT, sha256 TEXT, path TEXT,
  state INTEGER NOT NULL DEFAULT 0,        -- 0未下载 1下载中 2就绪 3损坏
  installed_at INTEGER
);
```

### 2.7 活动、归档、任务

```sql
CREATE TABLE activity (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  ts INTEGER NOT NULL,
  kind INTEGER NOT NULL,                   -- 1view 2create 3edit 4search 5clip 6archive_view
  entry_id TEXT, dwell_ms INTEGER, query TEXT, extra TEXT
);
CREATE INDEX idx_activity_ts ON activity(ts);

CREATE TABLE blob_ref (
  entry_id TEXT NOT NULL, blob_hash TEXT NOT NULL,
  kind INTEGER NOT NULL,                   -- 1text 2pdf 3html 4image 5favicon
  bytes INTEGER NOT NULL,
  local INTEGER NOT NULL DEFAULT 0,        -- 0仅云端 1本地已有
  created_at INTEGER NOT NULL,
  PRIMARY KEY (entry_id, blob_hash)
);
CREATE INDEX idx_blob_local ON blob_ref(local, bytes);

CREATE TABLE link_check (
  entry_id TEXT NOT NULL, checked_at INTEGER NOT NULL,
  http_status INTEGER,
  result TEXT NOT NULL,                    -- ok|dead|restricted|unreachable|changed
  final_url TEXT, simhash TEXT,
  PRIMARY KEY (entry_id, checked_at)
);

CREATE TABLE job_queue (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  type TEXT NOT NULL,                      -- embed|archive|reembed_all|deadlink
  payload TEXT,
  state INTEGER NOT NULL DEFAULT 0,        -- 0pending 1running 2done 3failed
  attempts INTEGER NOT NULL DEFAULT 0,
  next_run_at INTEGER NOT NULL DEFAULT 0,
  created_at INTEGER NOT NULL
);
CREATE INDEX idx_job_ready ON job_queue(state, next_run_at);
```

## 3. `entry.meta` 结构（按 kind 区分）

所有扩展字段都进 `meta` JSON，**不给 entry 表加列**，避免迁移成本。

```jsonc
// kind=1/2 笔记 / 想法
{ "tags": ["webdav"], "pinned": false, "word_count": 1240 }

// kind=3/4 日记
{ "date": "2026-11-08", "weather": "阴", "mood": "平静",
  "word_count": 320, "backfilled": false,
  "granularity": null }          // AI 摘要类：week|month|year

// kind=4 AI 日记额外
{ "citations": ["entry-id-1", "entry-id-2"], "model": "on-device" }

// kind=5 网页归档
{ "url": "...", "final_url": "...", "site": "example.com",
  "title": "...", "author": "...", "published_at": 1730000000000,
  "archive_status": "full",       // none|metadata|text|full|blocked|paywalled|dead
  "http_status": 200, "last_checked_at": 1731052800000,
  "recheck_policy": "monthly",    // weekly|monthly|never
  "snapshot_bytes": 412000, "reading_time_min": 8,
  "excerpt": "前 200 字", "changed": false, "wayback_url": null }

// kind=6 密码（业务字段在 credential 表，meta 只放轻量标记）
{ "category_hint": "工作", "has_totp": true }
```

## 4. `credential.custom_fields` 结构

```json
[
  { "id": "f1", "label": "恢复码", "value": "xxxx-xxxx-xxxx",
    "type": "text", "secret": true,  "sort": 0 },
  { "id": "f2", "label": "PIN",    "value": "8842",
    "type": "text", "secret": true,  "sort": 1 },
  { "id": "f3", "label": "服务器", "value": "sftp.example.com",
    "type": "url",  "secret": false, "sort": 2 }
]
```

`secret: true` 的字段与密码同级：默认掩码、复制后 30 秒清剪贴板、**绝不进入任何 prompt 上下文**。

## 5. 实体关系

```
entry ──1:1── credential
  │
  ├──N:M── category   （通过 entry_category，一条最多 1 个分类）
  ├──N:M── tag        （通过 entry_tag，一条最多 8 个）
  ├──1:N── entry_history
  ├──1:N── entry_vec  （按 chunk 分）
  ├──1:N── blob_ref
  └──1:N── link_check

category ──1:N── category（parent_id，两级）
category ──1:N── category_field_template

conversation ──1:N── message

oplog 是横切表：记录以上所有实体的变更
```

## 6. 同步策略表

| 实体 | 云端路径 | 合并策略 |
|---|---|---|
| entry | `items/<0-f>/<hmac(id)>.enc` | 笔记行级三方合并；书签/密码 LWW |
| category / tag | `taxonomy/<hmac(id)>.enc` | LWW + **同名合并**（取 usage 大的一方） |
| credential | 随 entry 一起（同文件内） | LWW + 保留旧值副本 |
| conversation / message | `chat/<hmac(id)>.enc` | 消息 append-only，会话 LWW |
| blob | `blobs/<sha256>.bin` | 内容寻址，不可变，无需合并 |
| vec | `vec/<hmac(id)>.enc` | model_id 不一致则丢弃重算 |
| meta | `meta.enc` | 全局单文件，**不用它做变更检测** |

## 7. 迁移与兼容

| 场景 | 规则 |
|---|---|
| 本地 schema 升级 | drift `MigrationStrategy`，每次迁移必须有单元测试 + 回滚测试 |
| 云端 `meta.schema` 高于本地 | **拒绝同步**，提示升级 App |
| `vault_id` 不匹配 | **拒绝同步**，弹窗让用户确认是否切换 vault |
| 索引表损坏 | 可从源数据全量重建（`entry_fts` / `entry_vec` / `diary_index`） |
| `oplog` 损坏 | 从 `entry_history` + `sync_state` 反推，最坏情况全量重同步 |

## 8. 容量估算（1 万条目 / 3 设备）

| 项 | 大小 |
|---|---|
| entry 表（含正文） | ~120 MB |
| entry_fts | ~60 MB |
| entry_vec（int8, 768 维） | ~8 MB |
| blob 元数据 | ~2 MB |
| oplog（保留 90 天） | ~40 MB |
| **本地合计** | **~230 MB** |
| 云端（加密，不含快照） | ~150 MB |
| 云端快照（1000 条 full） | ~1.2 GB |

> 快照是唯一会失控的部分，所以 `blob_ref.local` 默认 0（按需下载），并设配额守卫。
